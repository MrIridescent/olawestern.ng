2. Database Schema (Supabase PostgreSQL)






SQL
-- 1. Accomplishments & Track Record Table
CREATE TABLE accomplishments (
   id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   title TEXT NOT NULL,
   category TEXT NOT NULL CHECK (category IN ('Infrastructure', 'Youth & Sports', 'Security', 'Philanthropy', 'Legislative')),
   ward_impact TEXT NOT NULL, -- e.g., 'Festac II', 'Mile 2', 'All Wards'
   description TEXT NOT NULL,
   image_url TEXT,
   verification_source_url TEXT -- Link to news report or official press release
);

-- 2. Fact Check & Reputation Defense Table
CREATE TABLE fact_checks (
   id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   claim_summary TEXT NOT NULL,
   verdict TEXT NOT NULL CHECK (verdict IN ('VERIFIED FACT', 'FALSE CLAIM', 'MISLEADING')),
   detailed_explanation TEXT NOT NULL,
   official_document_url TEXT,
   is_featured BOOLEAN DEFAULT false
);

-- 3. Constituent Inquiries & Rumor Reports
CREATE TABLE constituent_inquiries (
   id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
   full_name TEXT NOT NULL,
   phone_number TEXT NOT NULL,
   ward TEXT NOT NULL,
   message_type TEXT CHECK (message_type IN ('Policy Inquiry', 'Rumor Verification', 'General Support')),
   message_body TEXT NOT NULL,
   status TEXT DEFAULT 'pending_review'
);

3. API Route Implementation: Truth Room & Fact-Check Fetching (app/api/fact-check/route.ts)






TypeScript
import { NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
 process.env.NEXT_PUBLIC_SUPABASE_URL!,
 process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
);

export async function GET() {
 try {
   const { data: factChecks, error } = await supabase
     .from('fact_checks')
     .select('*')
     .order('created_at', { ascending: false });

   if (error) throw error;

   return NextResponse.json({ success: true, data: factChecks });
 } catch (err: any) {
   return NextResponse.json({ success: false, error: err.message }, { status: 500 });
 }
}

4. API Route Implementation: Real-Time War Room Sync for Fact-Checking (app/api/truth-room/sync/route.ts)
When the AI Campaign War Room identifies a viral smear online, it posts an authenticated payload to this webhook, automatically generating a draft "Fact Check" post on the personal website for rapid executive approval:






TypeScript
import { NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
 process.env.SUPABASE_SERVICE_ROLE_KEY_URL!,
 process.env.SUPABASE_SERVICE_ROLE_KEY!
);

export async function POST(request: Request) {
 try {
   const authHeader = request.headers.get('authorization');
   if (authHeader !== `Bearer ${process.env.WAR_ROOM_SECRET_KEY}`) {
     return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
   }

   const { claimSummary, verdict, detailedExplanation, verificationUrl } = await request.json();

   const { data, error } = await supabase
     .from('fact_checks')
     .insert([
       {
         claim_summary: claimSummary,
         verdict: verdict,
         detailed_explanation: detailedExplanation,
         official_document_url: verificationUrl,
         is_featured: true
       }
     ]);

   if (error) throw error;

   return NextResponse.json({ success: true, message: 'Truth Room updated from War Room!' });
 } catch (err: any) {
   return NextResponse.json({ success: false, error: err.message }, { status: 500 });
 }
}
